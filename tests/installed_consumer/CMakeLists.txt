cmake_minimum_required(VERSION 3.24)
project(sam3d_installed_consumer LANGUAGES C)
find_package(sam3d CONFIG REQUIRED)
option(CONSUMER_SANITIZERS "Match the installed library's sanitizer runtimes" OFF)
add_executable(consumer ../test_body_model_api.c)
target_link_libraries(consumer PRIVATE sam3d::sam3d)
if(CONSUMER_SANITIZERS)
  target_compile_options(consumer PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(consumer PRIVATE -fsanitize=address,undefined)
endif()
