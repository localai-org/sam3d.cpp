option(SAM3D_SANITIZERS "Enable address/undefined-behavior sanitizers" ON)

option(SAM3D_FUZZ "Build the sanitizer-enabled C API libFuzzer target" OFF)
option(SAM3D_VULKAN "Build the dynamically loaded GGML Vulkan backend" OFF)
option(SAM3D_EXPERIMENTAL_GGML_PATCHES "Apply experimental GGML patches to a build-tree copy" OFF)

if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/ggml/CMakeLists.txt")
  message(FATAL_ERROR "GGML submodule missing: run git submodule update --init --recursive")
endif()
set(BUILD_SHARED_LIBS ON CACHE BOOL "Build shared dependencies" FORCE)
set(GGML_BACKEND_DL ON CACHE BOOL "Load explicit GGML backend modules" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "Native-only CPU flags conflict with dynamic backend loading" FORCE)
set(GGML_CPU ON CACHE BOOL "Build CPU backend" FORCE)
set(GGML_VULKAN ${SAM3D_VULKAN} CACHE BOOL "Build Vulkan backend" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "GGML upstream tests" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "GGML upstream examples" FORCE)
if(SAM3D_SANITIZERS AND SAM3D_ADDRESS_SANITIZER)
  set(GGML_SANITIZE_ADDRESS ON CACHE BOOL "GGML ASan" FORCE)
else()
  set(GGML_SANITIZE_ADDRESS OFF CACHE BOOL "GGML ASan" FORCE)
endif()
set(GGML_SANITIZE_UNDEFINED ${SAM3D_SANITIZERS} CACHE BOOL "GGML UBSan" FORCE)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(SAM3D_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  # A UBSan finding must fail tests, including GGML operations.
  add_compile_options(-fno-sanitize-recover=all)
endif()
if(UNIX AND NOT APPLE)
  set(CMAKE_INSTALL_RPATH "$ORIGIN;$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
endif()
if(SAM3D_VULKAN)
  find_package(SPIRV-Headers CONFIG REQUIRED)
endif()
if(SAM3D_EXPERIMENTAL_GGML_PATCHES)
  file(GLOB sam3d_ggml_patches CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/patches/ggml/*.patch")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${sam3d_ggml_patches})
  include(cmake/PrepareGGML.cmake)
  sam3d_prepare_ggml(sam3d_ggml_source)
  # ExternalProject's shader-generator cache also identifies its source path.
  # Give each source fingerprint a matching binary tree, including that cache.
  get_filename_component(sam3d_ggml_fingerprint_dir "${sam3d_ggml_source}" NAME)
  add_subdirectory("${sam3d_ggml_source}" "_deps/${sam3d_ggml_fingerprint_dir}-build")
else()
  add_subdirectory(ggml)
endif()
if(SAM3D_VULKAN)
  # GGML 0.23 finds this package but does not attach its include interface.
  # Needed for split-prefix SDK installations; no upstream source mutation.
  target_link_libraries(ggml-vulkan PRIVATE SPIRV-Headers::SPIRV-Headers)
endif()

    message(FATAL_ERROR "SAM3D_SANITIZERS requires GCC or Clang")

    target_link_options(sam3d_sanitizers INTERFACE -fsanitize=address)
  endif()
